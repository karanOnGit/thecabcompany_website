// src/app/components/BlogModal.tsx
'use client';

import { useState, useEffect, useRef } from 'react';
import {
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Button,
  TextField,
  MenuItem,
  Box,
  FormControl,
  InputLabel,
  Select,
  Alert,
  CircularProgress,
  Typography,
  Switch,
  FormControlLabel,
} from '@mui/material';
import ReactQuill from 'react-quill-new';
import 'react-quill-new/dist/quill.snow.css';
import { Blog } from '../types/blog';



// Correct import — capital B


const statuses = ['published', 'draft'] as const;
const visibilities = ['public', 'private'] as const;
const categories = [
  'News',
  'Reviews',
  'Tips & Tricks',
  'Technology',
  'Events',
  'Comparisons',
  'Guides',
];

interface BlogModalProps {
  open: boolean;
  onClose: () => void;
  blog: Blog | null;
  onSuccess: () => void;
}

interface FormData {
  title: string;
  slug: string;
  meta_title: string;
  meta_description: string;
  meta_keyword: string;
  tags: string;
  short_description: string;
  author: string;
  category: string;
  blog_image: string;
  detail_description: string;
  status: 'published' | 'draft';
  visibility: 'public' | 'private';
  pinned: boolean;
  slugEdited: boolean;
}

const generateSlug = (text: string): string =>
  text
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');

export default function BlogModal({
  open,
  onClose,
  blog,
  onSuccess,
}: BlogModalProps) {
  const quillRef = useRef<ReactQuill>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const [formData, setFormData] = useState<FormData>({
    title: '',
    slug: '',
    meta_title: '',
    meta_description: '',
    meta_keyword: '',
    tags: '',
    short_description: '',
    author: '',
    category: '',
    blog_image: '',
    detail_description: '',
    status: 'published',
    visibility: 'public',
    pinned: false,
    slugEdited: false,
  });

  useEffect(() => {
    if (blog) {
      setFormData({
        title: blog.title ?? '',
        slug: blog.slug ?? '',
        meta_title: blog.meta_title ?? '',
        meta_description: blog.meta_description ?? '',
        meta_keyword: (blog.meta_keyword ?? []).join(', '),
        tags: (blog.tags ?? []).join(', '),
        short_description: blog.short_description ?? '',
        author: blog.author ?? '',
        category: blog.category ?? '',
        blog_image: blog.blog_image ?? '',
        detail_description: blog.body_text ?? blog.detail_description ?? '',
        status: blog.status ?? 'published',
        visibility: blog.visibility ?? 'public',
        pinned: blog.pinned ?? false,
        slugEdited: true,
      });

      setTimeout(() => {
        const content = blog.body_text ?? blog.detail_description ?? '';
        quillRef.current?.getEditor().clipboard.dangerouslyPasteHTML(content);
      }, 0);
    } else {
      setFormData({
        title: '',
        slug: '',
        meta_title: '',
        meta_description: '',
        meta_keyword: '',
        tags: '',
        short_description: '',
        author: '',
        category: '',
        blog_image: '',
        detail_description: '',
        status: 'published',
        visibility: 'public',
        pinned: false,
        slugEdited: false,
      });
    }
    setError(null);
  }, [blog, open]);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setError(null);

    const payload = {
      ...formData,
      meta_keyword: formData.meta_keyword
        .split(',')
        .map((s) => s.trim())
        .filter(Boolean),
      tags: formData.tags
        .split(',')
        .map((s) => s.trim())
        .filter(Boolean),
      body_text: formData.detail_description,
      detail_description: formData.detail_description,
      category: formData.category || undefined,
    };

    try {
      const url = blog?.id
        ? `https://api.carsnbike.com/api/blog/${blog.id}`
        : 'https://api.carsnbike.com/api/blog/';

      const res = await fetch(url, {
        method: blog?.id ? 'PUT' : 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      const json = await res.json();
      if (!res.ok) throw new Error(json.message || 'Failed to save');

      alert(blog?.id ? 'Blog updated!' : 'Blog created!');
      onSuccess();
      onClose();
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    } finally {
      setLoading(false);
    }
  };

  const handleImageError = (e: React.SyntheticEvent<HTMLImageElement>) => {
    e.currentTarget.style.display = 'none';
  };

  return (
    <Dialog open={open} onClose={onClose} maxWidth="md" fullWidth>
      <DialogTitle>{blog ? 'Edit Blog' : 'Create Blog'}</DialogTitle>

      {formData.blog_image && (
        <Box display="flex" justifyContent="center" sx={{ width: '100%', pt: 2 }}>
          <Box
            component="img"
            src={formData.blog_image}
            alt="Preview"
            onError={handleImageError}
            sx={{
              width: '100%',
              maxHeight: 250,
              objectFit: 'cover',
              borderRadius: 1,
              border: '1px solid #ddd',
            }}
          />
        </Box>
      )}

      <Box component="form" onSubmit={handleSubmit}>
        <DialogContent dividers>
          {error && <Alert severity="error" sx={{ mb: 2 }}>{error}</Alert>}

          <Box display="grid" gap={2} gridTemplateColumns="1fr 1fr">
            <TextField
              label="Title"
              value={formData.title}
              onChange={(e) => {
                const value = e.target.value;
                setFormData((prev) => ({
                  ...prev,
                  title: value,
                  slug: prev.slugEdited ? prev.slug : generateSlug(value),
                }));
              }}
              required
            />
            <TextField
              label="Slug (editable)"
              value={formData.slug}
              onChange={(e) => {
                const value = e.target.value;
                setFormData((prev) => ({
                  ...prev,
                  slug: generateSlug(value),
                  slugEdited: true,
                }));
              }}
              required
            />
          </Box>

          <Box mt={2} display="grid" gap={2} gridTemplateColumns="1fr 1fr">
            <TextField label="Meta Title" value={formData.meta_title} onChange={(e) => setFormData((p) => ({ ...p, meta_title: e.target.value }))} />
            <TextField label="Meta Keywords (comma separated)" value={formData.meta_keyword} onChange={(e) => setFormData((p) => ({ ...p, meta_keyword: e.target.value }))} />
            <TextField label="Tags (comma separated)" value={formData.tags} onChange={(e) => setFormData((p) => ({ ...p, tags: e.target.value }))} />
            <TextField label="Author" value={formData.author} onChange={(e) => setFormData((p) => ({ ...p, author: e.target.value }))} required />
          </Box>

          <Box mt={2}>
            <TextField
              label="Meta Description"
              value={formData.meta_description}
              onChange={(e) => setFormData((p) => ({ ...p, meta_description: e.target.value }))}
              fullWidth
              multiline
              rows={2}
            />
          </Box>

          <Box mt={2}>
            <TextField
              label="Short Description"
              value={formData.short_description}
              onChange={(e) => setFormData((p) => ({ ...p, short_description: e.target.value }))}
              required
              fullWidth
              multiline
              rows={2}
            />
          </Box>

          <Box mt={2}>
            <TextField
              label="Blog Image URL"
              value={formData.blog_image}
              onChange={(e) => setFormData((p) => ({ ...p, blog_image: e.target.value }))}
              required
              fullWidth
            />
          </Box>

          <Box mt={2}>
            <FormControl fullWidth>
              <InputLabel>Category</InputLabel>
              <Select
                value={formData.category}
                onChange={(e) => setFormData((p) => ({ ...p, category: e.target.value }))}
              >
                {categories.map((c) => (
                  <MenuItem key={c} value={c}>{c}</MenuItem>
                ))}
              </Select>
            </FormControl>
          </Box>

          <Box display="flex" gap={2} mt={2}>
            <FormControl fullWidth>
              <InputLabel>Status</InputLabel>
              <Select
                value={formData.status}
                onChange={(e) => setFormData((p) => ({ ...p, status: e.target.value as 'published' | 'draft' }))}
              >
                {statuses.map((s) => (
                  <MenuItem key={s} value={s}>{s}</MenuItem>
                ))}
              </Select>
            </FormControl>

            <FormControl fullWidth>
              <InputLabel>Visibility</InputLabel>
              <Select
                value={formData.visibility}
                onChange={(e) => setFormData((p) => ({ ...p, visibility: e.target.value as 'public' | 'private' }))}
              >
                {visibilities.map((v) => (
                  <MenuItem key={v} value={v}>{v}</MenuItem>
                ))}
              </Select>
            </FormControl>
          </Box>

          <FormControlLabel
            control={<Switch checked={formData.pinned} onChange={(e) => setFormData((p) => ({ ...p, pinned: e.target.checked }))} />}
            label="Pinned Blog"
            sx={{ mt: 2 }}
          />

          <Typography variant="subtitle2" sx={{ mt: 3 }}>Blog Content</Typography>
          <ReactQuill
            ref={quillRef}
            theme="snow"
            value={formData.detail_description}
            onChange={(value) => setFormData((p) => ({ ...p, detail_description: value }))}
            style={{ height: 200, marginBottom: 40 }}
          />
        </DialogContent>

        <DialogActions>
          <Button onClick={onClose} disabled={loading}>Cancel</Button>
          <Button type="submit" variant="contained" disabled={loading}>
            {loading ? <CircularProgress size={20} /> : blog ? 'Update' : 'Create'}
          </Button>
        </DialogActions>
      </Box>
    </Dialog>
  );
}
